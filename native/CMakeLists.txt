cmake_minimum_required(VERSION 3.16)
project(RealtimeCSGRecovery LANGUAGES CXX)
find_package(Threads REQUIRED)

# Research module, not yet a replacement for the Unity native plugin.
add_library(realtimecsg_recovered_geometry SHARED
    recovered/geometry.cpp recovered/splitting.cpp recovered/categorization.cpp
    recovered/convex_csg.cpp recovered/convex_mesh_preparation.cpp recovered/convex_gap_repair.cpp
    recovered/scene_export.cpp recovered/scene_build.cpp recovered/scene_api.cpp
    recovered/triangulation.cpp recovered/mesh_creation.cpp recovered/mesh_cleanup.cpp
    recovered/mesh_optimization.cpp recovered/two_surface_candidates.cpp recovered/two_surface_decisions.cpp
    recovered/concave_triangulation.cpp recovered/plane_vertex_removal.cpp recovered/plane_vertex_pass.cpp recovered/two_surface_pass.cpp
    recovered/triangle_splitting.cpp recovered/gap_intervals.cpp recovered/gap_neighbors.cpp
    recovered/gap_dispatch.cpp recovered/brush_edges.cpp recovered/connections.cpp recovered/brush_edge_cleanup.cpp)
target_include_directories(realtimecsg_recovered_geometry PUBLIC include)
target_compile_features(realtimecsg_recovered_geometry PUBLIC cxx_std_17)
target_link_libraries(realtimecsg_recovered_geometry PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(realtimecsg_recovered_geometry PRIVATE /W4 /fp:strict)
    set_target_properties(realtimecsg_recovered_geometry PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
    target_compile_options(realtimecsg_recovered_geometry PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
